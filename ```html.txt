<script>

let lane=1;
let score=0;
let best=localStorage.getItem("best") || 0;
let gameOver=false;
let speed=5;
let skateboard=false;

const game=document.getElementById("game");
const player=document.getElementById("player");
const skate=document.getElementById("skate");
const scoreText=document.getElementById("score");
const bestText=document.getElementById("best");

bestText.textContent=best;

const lanes=[
  "20%",
  "50%",
  "80%"
];

function setLane(){
  player.style.left=lanes[lane];
  skate.style.left=lanes[lane];
}

setLane();

/* MOVE */
function move(dir){
  if(gameOver)return;

  lane+=dir;

  if(lane<0)lane=0;
  if(lane>2)lane=2;

  setLane();
}

/* JUMP */
function jump(){
  if(gameOver)return;

  player.classList.remove("jump");
  void player.offsetWidth;
  player.classList.add("jump");
}

/* KEYBOARD */
document.addEventListener("keydown",e=>{
  if(e.key==="ArrowLeft" || e.key.toLowerCase()==="a")move(-1);
  if(e.key==="ArrowRight" || e.key.toLowerCase()==="d")move(1);
  if(e.key==="ArrowUp" || e.code==="Space")jump();
});

/* CREATE BUS */
function createBus(){

  if(gameOver)return;

  const bus=document.createElement("div");
  bus.className="bus";

  let busLane=Math.floor(Math.random()*3);
  bus.dataset.lane=busLane;

  bus.style.left=lanes[busLane];
  bus.style.top="-100px";
  bus.style.transform="translateX(-50%)";

  game.appendChild(bus);

  let y=-100;

  function animate(){

    if(gameOver){
      bus.remove();
      return;
    }

    y+=speed;
    bus.style.top=y+"px";

    if(y>window.innerHeight){
      bus.remove();
      return;
    }

    /* collision */
    const playerRect=player.getBoundingClientRect();
    const busRect=bus.getBoundingClientRect();

    if(
      playerRect.left<busRect.right &&
      playerRect.right>busRect.left &&
      playerRect.top<busRect.bottom &&
      playerRect.bottom>busRect.top
    ){

      if(!player.classList.contains("jump")){
        crash();
        return;
      }
    }

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
}

/* CREATE STAR */
function createStar(){

  if(gameOver)return;

  const star=document.createElement("div");
  star.className="star";
  star.textContent="⭐";

  let starLane=Math.floor(Math.random()*3);

  star.style.left=lanes[starLane];
  star.style.top="-60px";
  star.style.transform="translateX(-50%)";

  game.appendChild(star);

  let y=-60;

  function animate(){

    if(gameOver){
      star.remove();
      return;
    }

    y+=speed;
    star.style.top=y+"px";

    const p=player.getBoundingClientRect();
    const s=star.getBoundingClientRect();

    if(
      p.left<s.right &&
      p.right>s.left &&
      p.top<s.bottom &&
      p.bottom>s.top
    ){

      score++;
      scoreText.textContent=score;

      if(score>best){
        best=score;
        bestText.textContent=best;
        localStorage.setItem("best",best);
      }

      star.remove();
      return;
    }

    if(y>window.innerHeight){
      star.remove();
      return;
    }

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
}

/* SKATEBOARD POWERUP */
function createSkate(){

  if(gameOver)return;

  const power=document.createElement("div");

  power.textContent="🛹";
  power.style.position="absolute";
  power.style.fontSize="48px";
  power.style.left=lanes[Math.floor(Math.random()*3)];
  power.style.top="-60px";
  power.style.zIndex="9";

  game.appendChild(power);

  let y=-60;

  function animate(){

    if(gameOver){
      power.remove();
      return;
    }

    y+=speed;
    power.style.top=y+"px";

    const p=player.getBoundingClientRect();
    const s=power.getBoundingClientRect();

    if(
      p.left<s.right &&
      p.right>s.left &&
      p.top<s.bottom &&
      p.bottom>s.top
    ){
      activateSkateboard();
      power.remove();
      return;
    }

    if(y>window.innerHeight){
      power.remove();
      return;
    }

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
}

function activateSkateboard(){

  skateboard=true;
  game.classList.add("skateMode");

  speed=9;

  setTimeout(()=>{
    skateboard=false;
    game.classList.remove("skateMode");
    speed=5;
  },6000);
}

/* CRASH */
function crash(){

  gameOver=true;

  document.getElementById("flash").classList.add("flash");

  player.style.transform="translateX(-50%) rotate(90deg)";

  document.getElementById("restartBtn").style.display="block";

  setTimeout(()=>{
    document.getElementById("flash").classList.remove("flash");
  },300);
}

/* GAME LOOP */
setInterval(()=>{
  if(!gameOver)createBus();
},1300);

setInterval(()=>{
  if(!gameOver)createStar();
},700);

setInterval(()=>{
  if(!gameOver && Math.random()<.5)createSkate();
},7000);

/* SCORE INCREASE */
setInterval(()=>{
  if(!gameOver){
    score++;
    scoreText.textContent=score;

    if(score>best){
      best=score;
      bestText.textContent=best;
      localStorage.setItem("best",best);
    }
  }
},2000);

/* SHOP */
document.getElementById("shopBtn").onclick=()=>{

};

function closeShop(){
  document.getElementById("shop").style.display="none";
}

function selectCharacter(type){

  const body=player.querySelector(".body");
  const head=player.querySelector(".head");

  if(type==="hulk"){
    body.style.background="#36a852";
    head.style.background="#54c45e";
  }

  if(type==="iron"){
    body.style.background="#c62828";
    head.style.background="#ddd";
  }

  if(type==="monkey"){
    body.style.background="#8d5524";
    head.textContent="🐒";
    head.style.background="transparent";
    head.style.border="0";
  }

  if(type==="king"){
    body.style.background="#7b1fa2";
    head.textContent="👑";
    head.style.background="transparent";
    head.style.border="0";
  }

  closeShop();
}

/* RESTART */
function restart(){
  location.reload();
}
