<script>

let lane = 1;
let score = 0;
let running = false;
let speed = 5;
let character = "💪";

let player = document.getElementById("player");
let game = document.getElementById("game");

let stars = [];
let buses = [];

let highScore = localStorage.getItem("highScore") || 0;
document.getElementById("highScore").innerText = highScore;


/* CHARACTER */
function selectCharacter(icon){

    character = icon;
    player.innerText = icon;
    document.getElementById("homeCharacter").innerText = icon;

    alert("Character Selected!");

    closeShop();
}

function buyCharacter(icon,cost){

    let currentStars = Number(localStorage.getItem("totalStars") || 0);

    if(currentStars >= cost){

        currentStars -= cost;

        localStorage.setItem("totalStars",currentStars);

        character = icon;
        player.innerText = icon;

        alert("Character Purchased!");

        updateShop();

    }else{

        alert("Not enough stars!");

    }
}

function updateShop(){

    let total = Number(localStorage.getItem("totalStars") || 0);

    document.getElementById("shopStars").innerText = total;
}


/* SHOP */
function openShop(){

    updateShop();

    document.getElementById("shop").style.display="block";
}

function closeShop(){

    document.getElementById("shop").style.display="none";
}


/* LANE */
function lanePosition(){

    let road = document.getElementById("road");

    let roadLeft = road.offsetLeft;
    let laneWidth = road.offsetWidth / 3;

    return roadLeft + laneWidth * lane + laneWidth/2 - 35;
}


/* MOVE */
function moveLeft(){

    if(!running)return;

    if(lane > 0){
        lane--;
        updatePlayer();
    }
}

function moveRight(){

    if(!running)return;

    if(lane < 2){
        lane++;
        updatePlayer();
    }
}

function updatePlayer(){

    player.style.left = lanePosition()+"px";

}


/* JUMP */
function jump(){

    if(!running)return;

    player.style.bottom="190px";

    setTimeout(()=>{
        player.style.bottom="80px";
    },500);
}


/* KEYBOARD */
document.addEventListener("keydown",function(e){

    if(e.key==="ArrowLeft" || e.key==="a"){
        moveLeft();
    }

    if(e.key==="ArrowRight" || e.key==="d"){
        moveRight();
    }

    if(e.key==="ArrowUp" || e.key==="w" || e.key===" "){
        jump();
    }

});


/* START GAME */
function startGame(){

    document.getElementById("home").style.display="none";
    document.getElementById("gameOver").style.display="none";

    running=true;
    score=0;
    lane=1;

    document.getElementById("score").innerText=score;

    updatePlayer();

    gameLoop();

}


/* CREATE BUS */
function createBus(){

    if(!running)return;

    let bus=document.createElement("div");

    bus.className="bus";

    let busLane=Math.floor(Math.random()*3);

    bus.dataset.lane=busLane;

    let road=document.getElementById("road");

    let roadLeft=road.offsetLeft;
    let laneWidth=road.offsetWidth/3;

    bus.style.left=
        (roadLeft + laneWidth*busLane + laneWidth/2 - 42)+"px";

    bus.style.top="-150px";

    game.appendChild(bus);

    buses.push(bus);

}


/* CREATE STAR */
function createStar(){

    if(!running)return;

    let star=document.createElement("div");

    star.className="star";
    star.innerText="⭐";

    let starLane=Math.floor(Math.random()*3);

    star.dataset.lane=starLane;

    let road=document.getElementById("road");

    let roadLeft=road.offsetLeft;
    let laneWidth=road.offsetWidth/3;

    star.style.left=
        (roadLeft + laneWidth*starLane + laneWidth/2 - 20)+"px";

    star.style.top="-50px";

    game.appendChild(star);

    stars.push(star);

}


/* COLLISION */
function collision(a,b){

    let r1=a.getBoundingClientRect();
    let r2=b.getBoundingClientRect();

    return !(
        r1.right < r2.left ||
        r1.left > r2.right ||
        r1.bottom < r2.top ||
        r1.top > r2.bottom
    );

}


/* GAME LOOP */
function gameLoop(){

    if(!running)return;

    /* BUSES */

    buses.forEach((bus,index)=>{

        let top=parseInt(bus.style.top);

        top+=speed;

        bus.style.top=top+"px";

        if(
            top>window.innerHeight
        ){

            bus.remove();
            buses.splice(index,1);

        }

        if(
            Number(bus.dataset.lane)===lane &&
            collision(player,bus)
        ){

            gameOver();

        }

    });


    /* STARS */

    stars.forEach((star,index)=>{

        let top=parseInt(star.style.top);

        top+=speed;

        star.style.top=top+"px";

        if(top>window.innerHeight){

            star.remove();
            stars.splice(index,1);

        }

        if(
            Number(star.dataset.lane)===lane &&
            collision(player,star)
        ){

            score++;

            document.getElementById("score").innerText=score;

            star.remove();
            stars.splice(index,1);

        }

    });


    requestAnimationFrame(gameLoop);

}


/* SPAWN */
setInterval(()=>{

    if(running){
        createBus();
    }

},1300);


setInterval(()=>{

    if(running){
        createStar();
    }

},700);


/* GAME OVER */
function gameOver(){

    if(!running)return;

    running=false;

    document.getElementById("finalScore").innerText=score;

    let oldHigh=Number(localStorage.getItem("highScore") || 0);

    if(score>oldHigh){

        localStorage.setItem("highScore",score);

        document.getElementById("highScore").innerText=score;

    }

    let totalStars=
        Number(localStorage.getItem("totalStars") || 0);

    localStorage.setItem(
        "totalStars",
        totalStars+score
    );

    document.getElementById("gameOver").style.display="flex";

}


/* RESTART */
function restartGame(){

    buses.forEach(x=>x.remove());
    stars.forEach(x=>x.remove());

    buses=[];
    stars=[];

    startGame();

}


/* HOME */
function goHome(){

    buses.forEach(x=>x.remove());
    stars.forEach(x=>x.remove());

    buses=[];
    stars=[];

    running=false;

    document.getElementById("gameOver").style.display="none";
    document.getElementById("home").style.display="flex";

}


/* INITIAL POSITION */
window.onload=function(){

    player.innerText=character;

    updatePlayer();

};
